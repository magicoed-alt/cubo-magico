// engine.js — cérebro do app "Cubo Mágico" (roda no JavaScriptCore do iPhone e no node para testes)
// Depende de: min2phase.js (solver Kociemba de 2 fases, cstimer) e cube.js
"use strict";

var Engine = (function () {
	var C = (typeof Cube !== 'undefined') ? Cube : require('./cube.txt');
	var M2P = (typeof min2phase !== 'undefined') ? min2phase : require('./min2phase.txt');

	// ---------------- configuração ----------------
	var DEF = {
		speechMode: 'posicao',          // 'posicao' | 'cores' | 'notacao'
		orientMode: 'auto',             // 'auto' = aprende pelos 2 primeiros giros (cima, direita) | 'fixa'
		faceNames: ['cima', 'direita', 'frente', 'baixo', 'esquerda', 'trás'],
		faceColor: ['branco', 'vermelho', 'verde', 'amarelo', 'laranja', 'azul'], // faces U R F D L B
		topFace: 0, frontFace: 2,       // como você segura o cubo (modos posição/notação)
		wordCW: 'frente', wordHalf: 'duplo', wordCCW: 'reverso',   // frente = horário olhando para a face
		steps: 1,                       // giros anunciados por vez
		repeatAfter: 3,                 // repetir a fala após X s sem giro (0 = nunca)
		maxRepeats: 3,
		waitAfterMix: 7,                // segundos sem giro para considerar a mistura terminada
		readyCue: 'bip', readyText: 'pronto',  // 'nada' | 'bip' | 'voz'
		doneCue: 'frase', doneText: 'ok',   // 'frase' = "the cube is solved" etc.
		optimizeMs: 800,                // tempo para procurar a solução mais curta
		echoMoves: false,               // fora de rotina: falar cada giro (teste de cores)
		loop: false,                    // ao terminar a playlist, recomeçar
		language: 'pt',                 // 'pt' | 'en' (frases fixas, cores e notação faladas)
		// como no CubeSmith: inverte o sentido falado das faces opostas, para girarem
		// no mesmo sentido que a direita / cima / frente
		invertLeft: false, invertBottom: false, invertBack: false
	};
	var cfg = clone(DEF);

	var POS = ['cima', 'direita', 'frente', 'baixo', 'esquerda', 'trás'];
	var LETTER = ['U', 'R', 'F', 'D', 'L', 'B'];
	var LETTER_SPEECH = ['u', 'érre', 'éfe', 'dê', 'éle', 'bê'];
	var TXT = {
		pt: { now: 'agora gire ', correct: 'corrige, ', no: 'não, ', error: 'erro', second: 'falta o segundo cubo',
			solved: 'o cubo está resolvido', matched: 'os cubos estão iguais', patternDone: 'padrão pronto',
			solveName: 'resolver o cubo', matchName: 'igualar os cubos',
			letters: LETTER_SPEECH, powers: ['', ' dois', ' linha'], patterns: {} },
		en: { now: 'now turn ', correct: 'correction, ', no: 'no, ', error: 'error', second: 'the second cube is missing',
			solved: 'the cube is solved', matched: 'the cubes match', patternDone: 'the pattern is ready',
			solveName: 'solve the cube', matchName: 'matching cubes',
			letters: ['U', 'R', 'F', 'D', 'L', 'B'], powers: ['', ' two', ' prime'],
			colors: { branco: 'white', vermelho: 'red', verde: 'green', amarelo: 'yellow', laranja: 'orange', azul: 'blue' },
			patterns: { 'Xadrez': 'checkerboard', 'Superflip': 'superflip', 'Cubo no cubo': 'cube in a cube',
				'Seis pontos': 'six spots', 'Faixas': 'bands', 'Cruzes': 'crosses', 'Listras': 'stripes' } }
	};
	function T() { return TXT[cfg.language] || TXT.pt; }
	function corFalada(c) { var t = T().colors; return (t && t[c]) || c; }
	var VEC = [[0, 1, 0], [1, 0, 0], [0, 0, 1], [0, -1, 0], [-1, 0, 0], [0, 0, -1]];

	function clone(o) { return JSON.parse(JSON.stringify(o)); }
	function dot(a, b) { return a[0] * b[0] + a[1] * b[1] + a[2] * b[2]; }
	function cross(a, b) { return [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]]; }

	// face interna (URFDLB do cubo) -> posição física conforme o cubo é segurado
	function usaOrientacaoAuto() { return cfg.orientMode == 'auto' && cfg.speechMode != 'cores'; }
	function physMap() {
		var up, fr, rt;
		if (run && run.orient && run.orient.top >= 0 && run.orient.right >= 0) {
			up = VEC[run.orient.top]; rt = VEC[run.orient.right]; fr = cross(rt, up);
		} else {
			var t = cfg.topFace | 0, f = cfg.frontFace | 0;
			if (t < 0 || t > 5 || f < 0 || f > 5 || dot(VEC[t], VEC[f]) != 0) { t = 0; f = 2; }
			up = VEC[t]; fr = VEC[f]; rt = cross(up, fr);
		}
		var map = [];
		for (var i = 0; i < 6; i++) {
			var v = VEC[i];
			if (dot(v, up) == 1) map[i] = 0; else if (dot(v, up) == -1) map[i] = 3;
			else if (dot(v, rt) == 1) map[i] = 1; else if (dot(v, rt) == -1) map[i] = 4;
			else if (dot(v, fr) == 1) map[i] = 2; else map[i] = 5;
		}
		return map;
	}

	function moveInfo(m) {
		var face = ~~(m / 3), p = m % 3, ph = physMap()[face];
		var pw = p;
		if (cfg.speechMode == 'posicao' && p != 1 &&
			((ph == 4 && cfg.invertLeft) || (ph == 3 && cfg.invertBottom) || (ph == 5 && cfg.invertBack))) pw = 2 - p;
		var w = [cfg.wordCW, cfg.wordHalf, cfg.wordCCW][pw];
		var color = cfg.faceColor[face] || C.FACES[face];
		var notation = LETTER[ph] + ['', '2', "'"][p];
		var speech;
		var pos = (cfg.faceNames && cfg.faceNames[ph]) || POS[ph];
		if (cfg.speechMode == 'posicao') speech = pos + ' ' + w;
		else if (cfg.speechMode == 'notacao') speech = T().letters[ph] + T().powers[p];
		else speech = corFalada(color) + ' ' + w;
		var nome = cfg.speechMode == 'cores' ? color : pos;
		return { m: m, face: face, power: p, speech: speech, notation: notation, color: color,
			label: nome.toUpperCase() + ' ' + ['↻', '2×', '↺'][p] };
	}

	// ---------------- solver ----------------
	var search = null, msPerProbe = 0.15, solverReady = false;
	function initSolver() {
		if (solverReady) return;
		M2P.initFull();
		search = new M2P.Search();
		solverReady = true;
	}
	function lenOf(s) { var t = s.trim(); return t ? t.split(/\s+/).length : 0; }
	function now() { return Date.now(); }

	// sequência de movimentos que leva o estado 'from' até 'to'
	function plan(from, to, budgetMs) {
		initSolver();
		var x = C.mult(C.inv(to), from);
		if (C.isSolved(x)) return [];
		var fac = C.toFacelet(x);
		var best = search.solution(fac, 21, 1e9, 0, 0);
		if (best.indexOf('Error') == 0) return null;
		var len = lenOf(best), t0 = now(), probe = 300;
		while (len > 1) {
			var left = budgetMs - (now() - t0);
			if (left <= 5) break;
			var pm = Math.min(probe, Math.max(50, Math.floor(left / msPerProbe)));
			var t1 = now();
			var r = search.solution(fac, len - 1, pm, 0, 0);
			var dt = now() - t1;
			if (search.probe > 20 && dt > 0) msPerProbe = msPerProbe * 0.5 + (dt / search.probe) * 0.5;
			if (r.indexOf('Error') == 0) {
				if (pm < probe) break;        // não deu tempo de procurar mais
				probe *= 3;
				if (probe > 3e6) break;
			} else { best = r; len = lenOf(r); }
		}
		return C.parseAlg(best);
	}

	// ---------------- cubos ----------------
	function newCube() {
		return { connected: false, raw: null, calibInv: new C.CC(), state: null, lastMoveAt: 0, lastMove: -1, moves: 0 };
	}
	var cubes = [newCube(), newCube()];
	var lastMoveAny = 0;
	var viewOrient = null;   // última orientação aprendida (para desenhar o cubo como ele está na mão)

	// ---------------- rotina ----------------
	var run = null;
	var out = [];
	function speak(text) {
		if (!text) return;
		var u = out.length ? out[out.length - 1] : null;
		if (u && u.type == 'speak') u.text += '. ' + text;   // uma fala nova cortaria a anterior
		else out.push({ type: 'speak', text: text });
	}
	function cue(kind, text) {
		if (kind == 'bip') out.push({ type: 'cue' });
		else if (kind == 'voz') speak(text);
	}

	function faceNome(i) { return (cfg.faceNames && cfg.faceNames[i]) || POS[i]; }
	function textoOrient1() { return faceNome(0) + ' ' + cfg.wordCW; }
	function textoOrient2() { return faceNome(1) + ' ' + cfg.wordCW; }
	function instrucao(text, t) {
		run.lastText = text; run.lastAct = t; run.repeats = 0;
		speak(text);
	}
	function eixo(f) { return f % 3; }

	var STEP_NAMES = { solve: 'Resolver', match: 'Igualar ao outro cubo', pattern: 'Padrão' };
	function stepName(st) {
		if (!st) return '';
		if (st.type == 'pattern') return 'Padrão: ' + (st.patternName || '?');
		return STEP_NAMES[st.type] || st.type;
	}
	// nome falado no começo da etapa ("solve the cube", "matching cubes", "prediction")
	function stepAnnounce(st) {
		if (!st) return '';
		if (st.announce && String(st.announce).trim()) return String(st.announce).trim();
		if (st.type == 'solve') return T().solveName;
		if (st.type == 'match') return T().matchName;
		var n = st.patternName || '';
		return T().patterns[n] || n;
	}
	function stepDone(st) {
		if (!st) return '';
		if (st.type == 'solve') return T().solved;
		if (st.type == 'match') return T().matched;
		return T().patternDone;
	}

	function startStep(t) {
		run.phase = 'wait';
		run.phaseStart = t;
		run.active = -1; run.plan = []; run.S = []; run.idx = 0; run.half = false;
		run.batchEnd = 0; run.lastAct = t; run.repeats = 0; run.message = '';
		run.orient = null; run.lastText = ''; run.halfRem = -1;
		speak(stepAnnounce(run.steps[run.stepIndex]));
	}

	function targetFor(st, k) {
		if (st.type == 'solve') return new C.CC();
		if (st.type == 'pattern') return C.fromFacelet(st.facelet) || new C.CC();
		if (st.type == 'match') { var o = cubes[1 - k]; return o.state ? o.state.clone() : null; }
		return new C.CC();
	}

	function buildPlan(t, budget, prefixo) {
		var st = run.steps[run.stepIndex];
		var cur = cubes[run.active].state;
		var target = targetFor(st, run.active);
		if (!target) { run.message = 'Segundo cubo não conectado'; return false; }
		run.target = target;
		var p = plan(cur, target, budget);
		if (!p) { run.message = 'Estado impossível (cubo descalibrado?)'; speak(T().error); return false; }
		run.plan = p;
		run.S = [cur];
		for (var i = 0; i < p.length; i++) run.S.push(C.applyMove(run.S[i], p[i]));
		run.idx = 0; run.half = false; run.halfRem = -1;
		if (p.length == 0) { finishStep(t); return true; }
		announce(t, prefixo);
		return true;
	}

	function announce(t, prefixo) {
		var n = Math.max(1, cfg.steps | 0);
		run.batchEnd = Math.min(run.idx + n, run.plan.length);
		run.lastText = batchText();
		speak((prefixo || '') + run.lastText);
		run.lastAct = t; run.repeats = 0;
	}
	// texto do(s) próximo(s) giro(s); no último giro começa com "now turn" / "agora gire"
	function batchText() {
		var parts = [];
		for (var i = run.idx; i < run.batchEnd; i++) {
			// meio giro feito: falta só um quarto de volta, no sentido em que ele começou
			var m = (i == run.idx && run.half && run.halfRem >= 0) ? run.halfRem : run.plan[i];
			parts.push(moveInfo(m).speech);
		}
		var txt = parts.join(', ');
		if (run.batchEnd >= run.plan.length && txt) txt = T().now + txt;
		return txt;
	}

	function finishStep(t) {
		var st = run.steps[run.stepIndex];
		if (cfg.doneCue == 'frase') speak(stepDone(st));
		else cue(cfg.doneCue, cfg.doneText);
		run.doneText = stepDone(st);
		run.stepIndex++;
		if (run.stepIndex >= run.steps.length) {
			if (cfg.loop) { run.stepIndex = 0; startStep(t); }
			else { run.phase = 'fim'; run.active = -1; run.plan = []; }
		} else startStep(t);
	}

	function onMove(k, t) {
		if (!run) {
			if (cfg.echoMoves && cubes[k].lastMove >= 0) speak(moveInfo(cubes[k].lastMove).speech);
			return;
		}
		if (run.phase == 'wait' || run.phase == 'fim') return;   // mistura em andamento
		var st = run.steps[run.stepIndex];
		if (run.phase == 'armed' || (run.phase == 'orient1' && k == run.active)) {
			if (st.type == 'match' && !cubes[1 - k].state) {
				run.message = 'Conecte o segundo cubo';
				speak(T().second);
				return;
			}
			run.active = k;
			if (usaOrientacaoAuto()) {
				// o giro que você deu foi na face de CIMA: agora o app sabe qual cor está em cima
				run.orient = { top: ~~(cubes[k].lastMove / 3), right: -1 };
				if (cubes[k].lastMove < 0) run.orient.top = -1;
				if (run.orient.top < 0) { instrucao(textoOrient1(), t); run.phase = 'orient1'; return; }
				run.phase = 'orient2';
				instrucao(textoOrient2(), t);
				return;
			}
			run.phase = 'guide';
			buildPlan(t, cfg.optimizeMs);
			return;
		}
		if (run.phase == 'orient1' || run.phase == 'orient2') {
			if (k != run.active) return;
			if (run.phase == 'orient2') {
				var f = cubes[k].lastMove >= 0 ? ~~(cubes[k].lastMove / 3) : -1;
				if (f < 0 || eixo(f) == eixo(run.orient.top)) {
					// girou cima ou baixo de novo: corrige
					instrucao(T().no + textoOrient2(), t);
					return;
				}
				run.orient.right = f;
				viewOrient = { top: run.orient.top, right: f };
				run.phase = 'guide';
				buildPlan(t, cfg.optimizeMs);
			}
			return;
		}
		if (run.phase != 'guide') return;
		if (k != run.active) {
			// mexeram no cubo-alvo durante o "igualar": recalcula
			if (st.type == 'match') buildPlan(t, 150);
			return;
		}
		var cur = cubes[k].state;
		if (C.equals(cur, run.target)) { finishStep(t); return; }
		var j = -1;
		for (var i = 0; i < run.S.length; i++) if (C.equals(cur, run.S[i])) { j = i; break; }
		if (j > run.idx) {
			run.idx = j; run.half = false; run.halfRem = -1; run.lastAct = t; run.repeats = 0;
			if (run.idx >= run.batchEnd) announce(t);
			else run.lastText = batchText();
			return;
		}
		if (j == run.idx) { run.half = false; run.halfRem = -1; run.lastAct = t; run.lastText = batchText(); return; }   // desfez o meio giro
		if (j >= 0) { run.idx = j; run.half = false; run.halfRem = -1; announce(t); return; }  // voltou atrás
		var m = run.plan[run.idx];
		if (m % 3 == 1) {
			// giro duplo: um quarto de volta (em qualquer sentido) é meio caminho
			var f3 = m - 1;
			var base = run.S[run.idx];
			var rem = C.equals(cur, C.applyMove(base, f3)) ? f3 : (C.equals(cur, C.applyMove(base, f3 + 2)) ? f3 + 2 : -1);
			if (rem >= 0) {
				run.half = true; run.halfRem = rem; run.lastAct = t; run.repeats = 0;
				run.lastText = batchText();
				return;
			}
		}
		// giro errado: recalcula a partir de onde está e avisa
		buildPlan(t, 150, T().correct);
	}

	function armar(t) {
		run.phase = 'armed';
		if (usaOrientacaoAuto()) {
			if (cfg.readyCue == 'bip') cue('bip');
			instrucao(textoOrient1(), t);    // "cima frente": o primeiro giro mostra qual face está em cima
		} else {
			run.lastText = '';
			cue(cfg.readyCue, cfg.readyText);
		}
	}

	function tick(t) {
		if (!run) return;
		if (run.phase == 'wait') {
			var ref = Math.max(lastMoveAny, run.phaseStart);
			if (t - ref >= cfg.waitAfterMix * 1000) armar(t);
		} else if (run.phase == 'guide' || run.phase == 'orient1' || run.phase == 'orient2' || (run.phase == 'armed' && run.lastText)) {
			if (cfg.repeatAfter > 0 && run.repeats < cfg.maxRepeats && t - run.lastAct >= cfg.repeatAfter * 1000) {
				speak(run.phase == 'guide' ? batchText() : run.lastText);
				run.repeats++; run.lastAct = t;
			}
		}
	}

	// ---------------- API para o app ----------------
	function take() { var o = out; out = []; return JSON.stringify(o); }

	function feed(k, bytes, t) {
		var p = C.giikerParse(bytes);
		if (!p) return take();
		var c = cubes[k];
		c.connected = true;
		c.raw = p.state;
		var st = C.mult(c.calibInv, p.state);
		var prev = c.state;
		c.state = st;
		if (prev && !C.equals(prev, st)) {
			c.lastMove = C.findMove(prev, st);
			c.lastMoveAt = t; c.moves++;
			lastMoveAny = t;
			onMove(k, t);
		}
		return take();
	}

	function cmd(name, arg, t) {
		arg = arg || {};
		switch (name) {
			case 'start':
				var steps = (arg.steps || []).filter(function (s) { return s && STEP_NAMES[s.type]; });
				if (!steps.length) break;
				run = { name: arg.name || 'Rotina', steps: steps, stepIndex: 0, doneText: '' };
				viewOrient = null;
				startStep(t);
				break;
			case 'stop': run = null; break;
			case 'tap':   // toque no óculos / fone: repete a instrução, ou pula a espera
				if (!run) break;
				if (run.phase == 'guide') { speak(batchText()); run.lastAct = t; }
				else if (run.phase == 'wait') armar(t);
				else if (run.lastText) { speak(run.lastText); run.lastAct = t; }
				else if (run.phase == 'armed') cue(cfg.readyCue, cfg.readyText);
				break;
			case 'reorient':   // você girou o cubo na mão: aprende de novo cima e direita
				if (run && run.active >= 0 && (run.phase == 'guide' || run.phase == 'orient2' || run.phase == 'orient1') && usaOrientacaoAuto()) {
					run.phase = 'orient1'; run.orient = null; run.plan = []; run.S = []; run.idx = 0;
					instrucao(textoOrient1(), t);
				} else if (run && run.phase != 'fim') startStep(t);
				break;
			case 'restartStep': if (run && run.phase != 'fim') startStep(t); break;
			case 'nextStep':
				if (run && run.phase != 'fim') { run.stepIndex++; if (run.stepIndex >= run.steps.length) run.phase = 'fim'; else startStep(t); }
				break;
			case 'calibrate':   // cubo está fisicamente resolvido agora
				var c = cubes[arg.slot | 0];
				if (c.raw) {
					c.calibInv = C.inv(c.raw);
					c.state = new C.CC();
					out.push({ type: 'calibrated', slot: arg.slot | 0, raw: C.toFacelet(c.raw) });
				}
				break;
			case 'setCalib':
				var cc = cubes[arg.slot | 0];
				var r = arg.raw ? C.fromFacelet(arg.raw) : null;
				cc.calibInv = r ? C.inv(r) : new C.CC();
				if (cc.raw) cc.state = C.mult(cc.calibInv, cc.raw);
				break;
			case 'capture':
				var cp = cubes[arg.slot | 0];
				if (cp.state) out.push({ type: 'captured', slot: arg.slot | 0, facelet: C.toFacelet(cp.state) });
				break;
			case 'connected':
				var cn = cubes[arg.slot | 0];
				cn.connected = !!arg.on;
				if (!arg.on) { cn.state = null; cn.raw = null; }
				break;
			case 'say':
				speak(arg.text);
				break;
			case 'testMove':
				speak(moveInfo(arg.m | 0).speech);
				break;
		}
		return take();
	}

	function config(json) {
		var o = typeof json == 'string' ? JSON.parse(json) : json;
		var n = clone(DEF);
		for (var k in o) if (o[k] !== null && o[k] !== undefined && k in DEF) n[k] = o[k];
		cfg = n;
	}

	function viewFaces() {
		if (viewOrient) return [viewOrient.top, viewOrient.right];
		var tp = cfg.topFace | 0, f = cfg.frontFace | 0;
		if (tp < 0 || tp > 5 || f < 0 || f > 5 || dot(VEC[tp], VEC[f]) != 0) { tp = 0; f = 2; }
		var rt = cross(VEC[tp], VEC[f]);
		for (var k = 0; k < 6; k++) if (dot(VEC[k], rt) == 1) return [tp, k];
		return [0, 1];
	}

	function displayText() {
		if (!run) return '';
		var st = run.steps[run.stepIndex];
		switch (run.phase) {
			case 'wait': return stepAnnounce(st);
			case 'armed': return run.lastText || stepAnnounce(st);
			case 'orient1': case 'orient2': return run.lastText;
			case 'guide': return batchText();
			case 'fim': return run.doneText || '';
		}
		return '';
	}

	function ui() {
		var u = { cubes: [], run: null, view: viewFaces() };
		for (var i = 0; i < 2; i++) {
			var c = cubes[i];
			u.cubes.push({
				connected: c.connected, hasState: !!c.state,
				facelet: c.state ? C.toFacelet(c.state) : '',
				solved: c.state ? C.isSolved(c.state) : false,
				lastMove: c.lastMove >= 0 ? moveInfo(c.lastMove).label + '  (' + moveInfo(c.lastMove).notation + ')' : '',
				moves: c.moves
			});
		}
		if (run) {
			var st = run.steps[run.stepIndex];
			var next = [];
			if (run.phase == 'guide') for (var j = run.idx; j < run.plan.length; j++) {
				var mi = moveInfo(run.plan[j]);
				next.push({ label: mi.label, notation: mi.notation, face: mi.face, power: mi.power, speech: mi.speech });
			}
			var labels = cfg.language == 'en' ? { wait: 'Waiting for the mix to end',
				armed: usaOrientacaoAuto() ? 'Ready — turn the TOP face of the cube you pick' : 'Ready — turn the cube you pick',
				orient1: 'Turn the TOP face', orient2: 'Turn the RIGHT face',
				guide: 'Guiding', fim: 'Routine finished' } : { wait: 'Aguardando a mistura terminar',
				armed: usaOrientacaoAuto() ? 'Pronto — gire a face de CIMA do cubo que você pegar' : 'Pronto — gire o cubo que você pegar',
				orient1: 'Gire a face de CIMA', orient2: 'Gire a face da DIREITA',
				guide: 'Guiando', fim: 'Rotina concluída' };
			u.run = {
				name: run.name, phase: run.phase, phaseLabel: labels[run.phase] || run.phase,
				stepIndex: run.stepIndex, stepCount: run.steps.length,
				stepName: run.phase == 'fim' ? '' : stepName(st),
				active: run.active, next: next, total: run.plan.length, done: run.idx,
				half: !!run.half, message: run.message || '',
				display: displayText(), last: run.phase == 'guide' && run.batchEnd >= run.plan.length,
				target: (run.phase == 'guide' && run.target) ? C.toFacelet(run.target) : ''
			};
		}
		return JSON.stringify(u);
	}

	// padrões prontos (aplicados a partir do cubo resolvido)
	var PATTERNS = [
		['Xadrez', "U2 D2 F2 B2 L2 R2"],
		['Superflip', "U R2 F B R B2 R U2 L B2 R U' D' R2 F R' L B2 U2 F2"],
		['Cubo no cubo', "F L F U' R U F2 L2 U' L' B D' B' L2 U"],
		['Seis pontos', "U D' R L' F B' U D'"],
		['Faixas', "U2 R2 L2 U2 R2 L2"],
		['Cruzes', "U F B' L2 U2 L2 F' B U2 L2 U"],
		['Listras', "F2 B2 U2 D2"]
	];
	function patterns() {
		return JSON.stringify(PATTERNS.map(function (p) {
			return { name: p[0], alg: p[1], facelet: C.toFacelet(C.applyAlg(new C.CC(), C.parseAlg(p[1]))) };
		}));
	}

	function moveLabels() {
		var r = [];
		for (var m = 0; m < 18; m++) r.push(moveInfo(m));
		return JSON.stringify(r);
	}

	return {
		config: config, feed: feed, tick: function (t) { tick(t); return take(); }, cmd: cmd, ui: ui,
		initSolver: initSolver, patterns: patterns, moveLabels: moveLabels, _plan: plan, _cubes: cubes
	};
})();

// Funções globais chamadas pelo Swift (JavaScriptCore)
function E_config(json) { Engine.config(json); return '[]'; }
function E_feed(slot, hex, t) {
	var b = [];
	for (var i = 0; i + 1 < hex.length; i += 2) b.push(parseInt(hex.substr(i, 2), 16));
	return Engine.feed(slot, b, t);
}
function E_tick(t) { return Engine.tick(t); }
function E_cmd(name, argJson, t) { return Engine.cmd(name, argJson ? JSON.parse(argJson) : {}, t); }
function E_ui() { return Engine.ui(); }
function E_init() { Engine.initSolver(); return 'ok'; }
function E_patterns() { return Engine.patterns(); }
function E_moves() { return Engine.moveLabels(); }
